```cpp
#pragma once
#include <cstdint>
#include <array>

namespace vp9 {

constexpr int kNumRefFrames = 8;
constexpr int kRefsPerFrame = 3;        // last, golden, altref
constexpr int kMaxSegments  = 8;
constexpr int kSegLvlMax    = 4;        // ALT_Q, ALT_LF, REF_FRAME, SKIP
constexpr int kMaxModeLfDeltas = 2;
constexpr int kMaxRefLfDeltas  = 4;

struct SegmentParams {
    uint8_t enabled;
    uint8_t update_map;
    uint8_t temporal_update;
    uint8_t update_data;
    uint8_t abs_delta;
    uint8_t tree_probs[7];
    uint8_t pred_probs[3];
    int16_t feature_data[kMaxSegments][kSegLvlMax]; // ALT_Q, ALT_LF (signed); REF, SKIP (binary)
    uint8_t feature_mask[kMaxSegments];             // bit i = feature i enabled
};

struct LoopFilterParams {
    uint8_t level;
    uint8_t sharpness;
    uint8_t mode_ref_delta_enabled;
    uint8_t mode_ref_delta_update;
    int8_t  ref_deltas[kMaxRefLfDeltas];   // INTRA, LAST, GOLDEN, ALTREF
    int8_t  mode_deltas[kMaxModeLfDeltas]; // ZERO, NEW
};

struct RefFrameDesc {
    uint8_t index;          // ref_frame_map[] slot (0..7); 0xFF = invalid
    uint8_t sign_bias;
};

struct PicParams {
    uint8_t  profile;
    uint8_t  bit_depth;          // 8 for Profile 0
    uint8_t  color_space;
    uint8_t  color_range;
    uint8_t  subsampling_x, subsampling_y;

    uint8_t  frame_type;         // 0 = KEY, 1 = INTER
    uint8_t  show_frame;
    uint8_t  show_existing_frame;
    uint8_t  show_existing_frame_idx;   // valid only if show_existing_frame
    uint8_t  error_resilient_mode;
    uint8_t  intra_only;
    uint8_t  allow_high_precision_mv;
    uint8_t  refresh_frame_context;
    uint8_t  frame_parallel_decoding_mode;
    uint8_t  reset_frame_context;
    uint8_t  frame_context_idx;         // 0..3

    uint32_t width, height;
    uint32_t render_width, render_height;

    // Refs: ref_frame_map[8] holds DPB slot indices; frame_refs[3] is the
    // per-frame {last, golden, altref} pick by name. Values 0..7 index
    // into ref_frame_map; 0xFF = unused.
    RefFrameDesc ref_frame_map[kNumRefFrames];
    RefFrameDesc frame_refs[kRefsPerFrame];

    uint8_t  refresh_frame_flags;       // bit i = update DPB slot i with this frame

    uint8_t  interp_filter;             // 0..3 fixed, 4 = SWITCHABLE
    uint8_t  txmode;                    // 0..3 fixed, 4 = TX_MODE_SELECT
    uint8_t  reference_mode;            // 0=single, 1=compound, 2=hybrid

    LoopFilterParams lf;
    SegmentParams    seg;

    uint8_t  base_qindex;
    int8_t   y_dc_delta_q;
    int8_t   uv_dc_delta_q;
    int8_t   uv_ac_delta_q;
    uint8_t  lossless;

    uint8_t  tile_cols_log2;
    uint8_t  tile_rows_log2;

    uint32_t header_size;               // compressed header size in bytes
    uint32_t uncompressed_header_size;  // for bitstream offset to compressed header
    uint32_t frame_size;                // total payload (incl. headers)
};

struct ProbUpdates {
    // Sparse — only fields the compressed header actually wrote.
    // Each "_present" flag means the regbuilder should patch the byte
    // offset in the prob buffer; "_value" carries the new value.
    uint8_t tx_mode_present;
    uint8_t tx_mode;

    // Coef probs: 4 tx sizes × 2 (intra/inter) × 2 (Y/UV) × 6 × 6 × 3.
    // Store as a packed bitmap of changed entries + values; regbuilder
    // walks the bitmap.
    uint8_t coef_present[4]; // per tx size
    // Compact representation; populated by parser, consumed by
    // regbuilder. See vp9_parser.cpp for exact layout.
    uint8_t coef_values[4 /*tx*/][2 /*ref*/][2 /*plane*/][6][6][3];
    uint8_t coef_changed[4][2][2][6][6][3];

    uint8_t skip_present;
    uint8_t skip[3];

    uint8_t inter_mode_present;
    uint8_t inter_mode[7][3];

    uint8_t interp_filter_present;
    uint8_t interp_filter[4][2];

    uint8_t is_inter_present;
    uint8_t is_inter[4];

    uint8_t ref_mode_present;     // single vs compound vs hybrid
    uint8_t reference_mode;       // when set, was selected by compressed header
    uint8_t comp_mode_present;
    uint8_t comp_mode[5];
    uint8_t single_ref_present;
    uint8_t single_ref[5][2];
    uint8_t comp_ref_present;
    uint8_t comp_ref[5];

    uint8_t y_mode_present;
    uint8_t y_mode[4][9];

    uint8_t partition_present;
    uint8_t partition[16][3];

    uint8_t mv_present;
    uint8_t mv_joints[3];
    uint8_t mv_sign[2];
    uint8_t mv_classes[2][10];
    uint8_t mv_class0[2];
    uint8_t mv_bits[2][10];
    uint8_t mv_class0_fp[2][2][3];
    uint8_t mv_fp[2][3];
    uint8_t mv_class0_hp[2];
```

```cpp
// mft/vp9_bool_decoder.h
#pragma once
#include <cstdint>
#include <cstddef>

namespace vp9 {

class BoolDecoder {
public:
    void init(const uint8_t *buf, size_t sz);
    int  decode_bool(uint8_t prob);            // spec read_bool(p)
    int  decode_literal(int bits);             // n successive prob=128 bools (MSB first)
    bool ok() const { return !error_; }
private:
    void fill();
    const uint8_t *buf_ = nullptr;
    size_t   sz_ = 0;
    size_t   pos_ = 0;
    uint32_t value_ = 0;
    uint32_t range_ = 255;
    int      bits_left_ = 0;
    bool     error_ = false;
};

} // namespace vp9
```

```cpp
// mft/vp9_bool_decoder.cpp — implements spec §9.2.1-9.2.2.
#include "vp9_bool_decoder.h"

namespace vp9 {

void BoolDecoder::init(const uint8_t *buf, size_t sz) {
    buf_ = buf; sz_ = sz; pos_ = 0;
    value_ = 0; range_ = 255; bits_left_ = 0; error_ = false;
    // Read the first marker bit (must be 0) plus initial 7 value bits.
    fill();
    if (decode_bool(128) != 0) error_ = true; // marker
}

void BoolDecoder::fill() {
    while (bits_left_ <= 0) {
        if (pos_ >= sz_) { error_ = true; return; }
        value_ = (value_ << 8) | buf_[pos_++];
        bits_left_ += 8;
    }
}

int BoolDecoder::decode_bool(uint8_t prob) {
    fill();
    if (error_) return 0;
    uint32_t split = 1 + (((range_ - 1) * prob) >> 8);
    uint32_t bigsplit = split << bits_left_;
    int bit;
    if (value_ >= bigsplit) {
        bit = 1;
        range_ -= split;
        value_ -= bigsplit;
    } else {
        bit = 0;
        range_ = split;
    }
    // Renormalize
    while (range_ < 128) {
        range_ <<= 1;
        value_ <<= 1;
        if (--bits_left_ <= 0) fill();
    }
    return bit;
}

int BoolDecoder::decode_literal(int bits) {
    int v = 0;
    for (int i = 0; i < bits; ++i) v = (v << 1) | decode_bool(128);
    return v;
}

} // namespace vp9
```

```cpp
int DecodeEngineVp9_DecodeOne(DecodeEngineVp9 *e, const uint8_t *frame,
                              size_t len, int *out_slot, int *out_show) {
    vp9::PicParams pp{}; vp9::ProbUpdates pu{};
    auto r = vp9::Vp9Parser_Parse(frame, len, e->parser_state, pp, pu);
    if (r != vp9::ParseResult::Ok) return -1;

    if (pp.show_existing_frame) {
        // No HW kick; emit cached slot directly.
        int slot = pp.show_existing_frame_idx; // already in DPB
        *out_slot = slot; *out_show = 1;
        return 0;
    }

    // Pick a free pool slot for the current frame.
    int slot = pick_free_pool_slot(e);
    if (slot < 0) return -2;
    e->cur_slot = slot;

    // Fill prob buffer for this frame_context_idx.
    uint8_t fcx = pp.frame_context_idx;
    vp9::Vp9Regbuilder_FillProbs(pp, pu,
        reinterpret_cast<uint8_t *>(e->prob[fcx].user_va));

    // Build dense job.
    vp9::RegbuildInputs in{};
    in.pp = &pp; in.dpb = &e->dpb;
    in.bitstream_handle = e->bitstream.cookie;
    in.bitstream_bytes  = static_cast<uint32_t>(len);
    in.bitstream_offset = 0;
    in.decout_frame_handle = e->pool_output[slot].cookie;
    in.decout_colmv_handle = e->pool_colmv[slot].cookie;
    in.prob_handle         = e->prob[fcx].cookie;
    in.prob_default_handle = e->prob_default.cookie;
    in.segid_last_handle   = e->segid[e->segid_phase].cookie;
    in.segid_cur_handle    = e->segid[e->segid_phase ^ 1].cookie;
    for (int i = 0; i < 10; ++i) in.rcb_handles[i] = e->rcb[i].cookie;
    in.error_ref_handle    = e->error_ref.cookie;
    in.last_intra_only     = e->last_intra_only;
    in.col_ref_poc         = e->col_ref_poc;
    in.segid_ref_poc       = e->segid_ref_poc;

    RKMPP_SUBMIT_DENSE_JOB_IN job{};
    if (vp9::Vp9Regbuilder_Fill(in, &job) != 0) return -3;

    // Copy bitstream payload to e->bitstream.user_va.
    memcpy(e->bitstream.user_va, frame, len);

    // Submit + wait via backend.
    uint64_t job_id = 0;
    if (e->backend->SubmitDense(e->backend->ctx, &job, &job_id) != 0) return -4;
    RKMPP_WAIT_JOB_OUT wo{};
    if (e->backend->WaitJob(e->backend->ctx, job_id, 1000, &wo) != 0) return -5;
    if (wo.Status != 0) return -6;

    // Apply DPB update + sticky state.
    vp9::Vp9Dpb_Update(e->dpb, pp, e->pool_output[slot].cookie,
                       e->pool_colmv[slot].cookie, pp.width, pp.height,
                       pp.bit_depth);
    vp9::Vp9Parser_ApplyDpbUpdate(e->parser_state, pp);
    e->last_intra_only = pp.intra_only || pp.frame_type == 0;
    if (pp.show_frame && !pp.show_existing_frame)
        e->col_ref_poc = e->dpb.next_poc - 1;
    if (pp.seg.update_map) e->segid_ref_poc = e->dpb.next_poc - 1;
    e->segid_phase ^= 1;

    *out_slot = slot; *out_show = pp.show_frame;
    return 0;
}
```

Expected on first try: most likely DEC_RDY with **wrong pixels**. Diagnostic order:
  1. Look at INT_STATUS: did DEC_RDY fire? If no → reg104/105 scaling math wrong (HW thinks frame size is bogus) or stream_len off.
